הקבצו ושמעו בני יעקב דת יוצרכם

by Judah Halevi c. 1075–1141

הִקָּבְצוּ וְשִׁמְעוּ בְּנֵי יַעֲקֹב דָּת יוֹצֶרְכֶם
לְחֻקּוֹתָיו וּלְתוֹרוֹתָיו הַטּוּ הַיּוֹם אָזְנֵיכֶם
כִּי לְעָבְדוֹ וּלְרוֹמְמוֹ אַתֶּם נִצָּבִים הַיּוֹם כֻּלְּכֶם
רָאשֵׁיכֶם שִׁבְטֵיכֶם זִקְנֵיכֶם וְשֹׁטְרֵיכֶם.

אַבִּיעָה חִידוֹת מִנִּי קֶדֶם אָז נִפְלָאוּ
אֲסַפְּרָה גְבוּרוֹתָיו בְּגוֹי עַם קָדוֹשׁ נִקְרָאוּ
חִישׁ יְחַדֵּשׁ לָכֶם אוֹתוֹתָיו וְנִסָּיו כְּחוֹזִים נִבָּאוּ
הָרִאשֹׁנוֹת הִנֵּה בָאוּ.

אֶשְׁאֲלָה רִשְׁיוֹן שֹׁכְנִי אַפַּדְנֵי מְרוֹמִים
אֲבָאֵר הַיּוֹם הִלְכוֹת פֶּסַח הַנְּעִימִים
אֲשֶׁר בֵּאֲרוּ רַבּוֹתֵינוּ בְּיֹשֶׁר וּבְתָמִים
וְהֵמָּה חֲכָמִים מְחֻכָּמִים.

בְּיֶרַח צֵאת בְּנוֹת אֵל מִיַּד חַמְסָן
בְּהַכּוֹתוֹ רֵאשִׁית כָּל בְּכוֹרֵי אַנְסָן
בּוֹ שַׁדַּי הִפְלִיא עֲלִילוֹת וַיָּרֶם קֶרֶן נִסָּן
בַּחֹדֶשׁ הָרִאשׁוֹן הוּא חֹדֶשׁ נִיסָן.

גֶּזַע עַם הַקֹּדֶשׁ הִזָּהֵר בְּנֹעַם אַזְהָרוֹת
גַּלֵּה אָזְנְךָ וְתִהְיֶינָה מַצּוֹת פֶּסַח שְׁמוּרוֹת
גַּם הִזָּהֵר לְבָרֵךְ עַל בִּעוּר חָמֵץ לְנֶאֱזָר בִּגְבוּרוֹת
בְּהַעֲלֹתְךָ אֶת הַנֵּרֹת.

דְּבָרוֹ שְׁמֹר אֲשֶׁר בְּכֶתֶם אוֹפִיר לֹא תְסֻלֶּה
דָּת בִּעוּר חָמֵץ לְאוֹר הַנֵּר עַד כַּלֵּה
דַּע וְהִזָּהֵר וּלְבִלְעֲדֵי אוֹר הַנֵּר – לֹא לְאוֹר הָאֲבוּקָה וְחַמָּה וּלְבָנָה – הֶחָמֵץ אַל תְּגַלֶּה
לֹא בָחַר יְיָ בָּאֵלֶּה.

הֵן בְּבָדְקְךָ אִם שִׁיַּרְתָּ כִּכְּרוֹת לֶחֶם בְּמִנְיָן אֲצוּלוֹת
הֹרוּ אִם חָסֵר מֵהֶם כְּלוּם שֶׁמָּא הֵם מִפִּי שֶׁרֶץ אֲכוּלוֹת
הֵם אָמְרוּ לִבְדֹּק שֵׁנִית וּמִסְתּוֹרִים לְגַלּוֹת
הָשֵׁב יָדְךָ כְּבוֹצֵר עַל סַלְסִלּוֹת.

וְאִם מָצָא פַּת בַּפֶּסַח חַג מְחוֹלַת מַחֲנַיִם
וְהוּא שֶׁכְּבָר בִּטֵּל בְּלִבּוֹ שֶׁלֹּא רָאָה בְּעַפְעַפַּיִם
וְאֵינוֹ נֶעֱנָשׁ כִּי רָאָה תֹּם לְבָבוֹ שׁוֹכֵן שָׁמַיִם
כִּי הָאָדָם יִרְאֶה לַעֵינַיִם.

זָוִיּוֹת חָצֵר וְאֶמְצָעִיִּים וְגַג יָצִיעַ וְרֶפֶת וְלוּלִים וּמַתְבֵּן וְאוֹצְרוֹת יַיִן וָשֶׁמֶן, חַכְמֵי עַם נְדָבוֹת
זָהֲרוּ לְבָדְקָם, וּבָתֵּי עֵצִים וְדוֹנַג וּמֶלַח וְתִימוֹרִים תִּבְדֹּק בְּצִוּוּי דָּת תּוֹכֵן לִבּוֹת
זוּלָתִי חוֹרֵי עֶלְיוֹנִים שֶׁגָּבְהוּ וְתַחְתּוֹנִים שֶׁעָמְקוּ וּלְבָדְקָם אֵינָם קְרוֹבוֹת
וְחָשְׁכוּ הָרֹאוֹת בָּאֲרֻבּוֹת.

חָצֵר תּוֹכוֹ אֵין אַתָּה צָרִיךְ לִבְדֹּק וְהָאֵל יָגֵן בַּעֲדֶךָ
חֹר שֶׁבֵּינְךָ לַאֲרַמַּאי אַל תִּבְדֹּק שֶׁמָּא בִּכְשָׁפִים יַחְשְׁדֶךָ
חֹר שֶׁבֵּינְךָ לְיִשְׂרָאֵל צִוּוּ תַּלְמִידֵי חֲסִידֶיךָ
וְעָשִׂיתָ לּוֹ כַּאֲשֶׁר תִּמְצָא יָדֶךָ.

טְהוֹרִים אָמְרוּ אִם נָפַל מַפֹּלֶת עַל חָמֵץ וִיכַסֶּנּוּ
טֶרֶם נָפְלוֹ אִם לֹא בְדָקוֹ נֶחְשָׁב כִּמְבֹעָר אֲשֶׁר בָּאֵשׁ תִּשְׂרְפֶנּוּ
טְפָחִים שְׁלֹשָׁה שֶׁיּוּכַל הַכֶּלֶב לְפַשְׁפֵּשׁ תַּחְפְּשֶׂנּוּ
וְעָמֹק עָמֹק מִי יִמְצָאֶנּוּ.

יִשְּׁרוּ בִּטּוּל חָמֵץ לְחֹדֶשׁ נִיסָן בְּאַרְבָּעָה עָשָׂר יוֹם
יִפּוּ לֶאֱכֹל כָּל אַרְבַּע וְלִתְלוֹת כָּל חָמֵשׁ וְלִשְׂרֹף בִּתְחִלַּת שֵׁשׁ לְהַמְצִיאָם פִּדְיוֹם
יְמַהֲרוּ בִּשְׁבִיל שֶׁמָּא יוֹם מְעֻנָּן וְיֶחֶטְאוּ לְאֵל נוֹרָא וְאָיֹם
וְהִתְמַהְמְהוּ עַד נְטוֹת הַיּוֹם.

כְּשֶׁתֵּצֵא לַדֶּרֶךְ אִם קֹדֶם הַפֶּסַח שְׁלֹשִׁים יוֹם וְתִנְעֹל בָּתֶּיךָ וְתִכְלָאֵם
כָּךְ אָמְרוּ שֶׁאַתָּה חַיָּב לְבַעֲרָם וּבְהִלְכוֹת פֶּסַח תִּקְרָאֵם
כָּל שְׂאוֹר וְכָל חָמֵץ שֶׁבְּיָדְךָ לַיָּם תַּטִּיל אוֹ בַמִּדְבָּר תְּאַבְּדֵם וּבִרְשׁוּתָךְ אַל תִּרְאֵם
תִּזְרֵם וְרוּחַ תִּשָּׂאֵם.

לֹוֶה אָדָם בַּדֶּרֶךְ חָמֵץ שֶׁבְּיָדוֹ לַגּוֹי בְּמִצְוָה נָאָה
לַעֲשׂוֹת מְחִצָּה בֵּינֵיהֶם אָמְרוּ, לְהַבְדִּילוֹ מֵעָוֹן וְחַטָּאָה
לָהֲגוּ חָמֵץ שֶׁהִכְנִיס גּוֹי אוֹ מְצָאוֹ בַּפֶּסַח לְכַסּוֹתוֹ בְּעֵת אוֹתוֹ רָאָה
בְּעַד הַשְּׁאֵרִית הַנִּמְצָאָה.

מַצּוֹת פֶּסַח אַל יוּלְשׁוּ בְּיַיִן וְשֶׁמֶן וּדְבַשׁ, וְאַל יַשְׁלִים אָדָם כָּל תַּאֲוָתוֹ
מִלְּפַלֵּחַ פִּתִּים בְּסִיר הַנָּזִיד הִשָּׁמֵר כְּמִצְוָתוֹ
מֻתָּר לַעֲשׂוֹתָם לְנַעַר וְזָקֵן וְחוֹלֶה בַּחֲלוֹתוֹ
בַּעֲבֻר אֲשֶׁר יְבָרֶכְךָ לִפְנֵי מוֹתוֹ.

נוֹדַע שֶׁחֹמֶץ הָאֲדֹמִי וְזִיתוֹם הַמִּצְרִי כְּמַעֲשֵׂה מִצְרַיִם
נֶאֶסְרוּ עִם כֻּתָּח הַבַּבְלִי וְשֵׁכָר הַמָּדִי לְזֶרַע יְהוּדָה וְאֶפְרַיִם
נָאֲמוּ שֶׁלֹּא יוּלְשׁוּ אֶלָּא בְּמַיָּא דְּבִיתוּ פֶּן יֵחַם עֲלֵיהֶם הַצָּהֳרַיִם
וְהַשֶּׁמֶשׁ זָרְחָה עַל הַמָּיִם.

סוֹדִי שִׁמְרוּ כַּאֲשֶׁר דִּבַּרְתִּי אֲלֵיכֶם
סַכִּינִין וּכְלֵי נְחֹשֶׁת לְהַגְעִיל בְּרוֹתְחִין וְלִמְרֹק כִּי כֵן נִתְחַיֵּב עֲלֵיכֶם
סוּרוּ מִבֶּצַע לְשַׁבֵּר כָּל כְּלֵי חֶרֶס וּלְהַרְחִיקָם מֵאָהֳלֵיכֶם
וְעֵינְכֶם אַל תָּחֹס עַל כְּלֵיכֶם.

עֵץ פָּארוּר שֶׁשִּׁמֵּשׁ בּוֹ הָאָדָם כָּל הַשָּׁנָה בְּמַבָּעוֹ
עָלָיו לְהַשְׁלִיךְ מַיִם חַמִּין דִּתְנַן כָּךְ פָּלְטוֹ כְּבַלְעוֹ
עֲרִיסַת חָמֵץ אִם יֵשׁ בּוֹ וְחֶלְאָתוֹ תִּתֹּם כַּאֲשֶׁר עִלְּעוֹ
וְהֹצֵאתִי אֶת בִּלְעוֹ.

פִּקּוּדֵי חֲכָמִים קַדְמוֹנִים בָּהֶם הִדָּבֵק כַּאֲשֶׁר פֵּרַשְׁתִּי
פִּסְחִי בְּלֶחֶם עֹנִי שְׁמֹר, וּבַעֲשׂוֹתְךָ כַּאֲשֶׁר הִזְהַרְתִּי
פִּתְחוֹן פֶּה יִהְיֶה לְךָ לֵאמֹר כִּי עֻנֵּיתִי וּמִפִּי מָנַעְתִּי
מַטְעַמִּים כַּאֲשֶׁר אָהַבְתִּי.

צְרִיכִין מַצּוֹת פֶּסַח שְׁנֵי כֵלִים מְצַנֶּנֶת וּמְקַטֶּפֶת וּבְיוֹמָם
צִוּוּ שֶׁלֹּא יָלִישׁוּ לַשֶּׁמֶשׁ פֶּן יִתְחַמְּצוּ הָעִסּוֹת בְּחֻמָּם
צִוּוּ שֶׁהַמַּצָּה מֵחֲמֵשֶׁת הַמִּינִים בְּשׁוּמָם
כָּכָה יֹאכְלוּ בְנֵי יִשְׂרָאֵל אֶת לַחְמָם.

קָרְאוּ שֶׁיִּתְעַסְּקוּ בָעִסָּה לָשָׁה וְעוֹרֶכֶת וְאוֹפָה בְּיֹפִי וּמַכְלוּל
קָטָן וְחֵרֵשׁ וְשׁוֹטֶה אַל יָלִישׁוּ בְּדֹפִי וְחִלּוּל
קִבּוּץ מִי הָעִסּוֹת יִשָּׁפְכוּ בְּמוֹרַד מִדְרוֹן מַסְלוּל
עַל הַר גָּבֹהַּ וְתָלוּל.

רָאוּי לִהְיוֹת הָעִסָּה מִשִּׁעוּר אַרְבָּעִים וְשָׁלֹשׁ בֵּיצִים וְחֹמֶשׁ בֵּיצָה כְּדָת פְּלִילִיָּה
רֵאשִׁית הָעִסָּה לְהַפְרִישׁ שְׁתֵּי חַלּוֹת לִתְרוּמִיָּה
רָחֲשׁוּ שֶׁלֹּא יִלְעֹס אָדָם חִטִּים לָשׂוּם עַל מַכָּתוֹ לְהָפִיק רְטָיָה
פֶּצַע וְחַבּוּרָה וּמַכָּה טְרִיָּה.

שִׁמְרוּ שֶׁתִּהְיֶינָה הַנָּשִׁים זְהִירוֹת כַּאֲנָשִׁים בַּעֲבוֹדָתָם
שִׁעְבּוּד עֳנִי אֲבוֹתֵינוּ לִזְכֹּר בִּהְיוֹתָם תַּחַת יַד מִצְרַיִם בְּסִבְלֹתָם
שַׁדַּי עָלֵינוּ לְשַׁבֵּחַ בְּזָכְרֵנוּ אֶת כָּל אוֹדוֹתָם
וְאֵת כָּל הַקֹּרֹת אֹתָם.

תְּבָרֵךְ לְשׁוֹנִי שֶׁהֶחֱיָנוּ וְקִיְּמָנוּ לְאֵל אָיֹם וְנוֹרָא
תָּמוֹךְ אֶתְמֹךְ פֶּסַח מַצָּה וּמָרוֹר וְאֶשְׁתֶּה שָׁלֹשׁ כּוֹסוֹת לִזְכִירָה
תְּהִלָּה אֶעֱרֹךְ בָּרְבִיעִי בַּאֲמִירָה טְהוֹרָה
כּוֹס יְשׁוּעוֹת אֶשָּׂא וּבְשֵׁם יְיָ אֶקְרָא.

אָסוּר לֶאֱכֹל כְּלוּם אַחַר כּוֹס הַהַלֵּל כְּמִצְוַת צְרוּפָה
נִצְּרֶהָ לְבַד אִם תִּצְמָא וְנַפְשְׁךָ בַּצָּמָא שְׁאוּפָה
יֻתַּן לְךָ מַיִם וְתִשְׁתֶּה לְהַמְצִיא לְצִמְאוֹנָךְ תְּרוּפָה
מַיִם קָרִים עַל נֶפֶשׁ עֲיֵפָה.

יֹשֶׁר הֲלָכוֹת וְצֶדֶק דִּבְרֵי הַתּוּשִׁיָּה
הַיּוֹם לָהַגְתִּי וָאֲסַפֵּר יְקָרְךָ בְּנֶפֶשׁ רְצוּיָה
שׁוֹבְבָנִי מְהֵרָה וְאָז אֲשׁוֹרֵר לְשִׁמְךָ רַב הָעֲלִילִיָּה
לַמּוֹעֵד הַזֶּה כָּעֵת חַיָּה.
