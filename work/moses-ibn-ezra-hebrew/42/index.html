Moses ibn Ezra: Hebrew poems (Wikisource) · 11th-12th century

ספר הענק, שער נ (Sefer ha-Anaq, chapter Nun)

by Moses ibn Ezra c. 1055–after 1138

קעח
רָאוּי לְכָל מַשְׂכִּיל לְהַכְנִיעַ
לִבּוֹ וְעִם יִצְרוֹ יְהִי נוֹכָח
וּבְעֵת בְּסֵתֶר יַחֲשֹׁב לַעְשׂוֹת
עָוֹן יְשַׁוֶּה יוֹצְרוֹ נוֹכַח.

קעט
קוֹל הַזְּמָן יַעְבִיר הֲכִי מִקְרֶה
חֹטֵא כְּמוֹ מִקְרֶה אֱנוֹשׁ נָקִיא
גַּם הָאֲרָצוֹת יִקְרְאוּ אֶל שֹׁכְ-
נֵיהֶם הֲמוֹנְכֶם מָחֳרָת נָקִיא.

קפ
אָמְנָם עֲבֹדַת הָאֱלֹהִים הִיא
מִסְעָד וּמָעֹז לֶאֱנוֹשׁ נוֹפֵל
הִיא לֶאֱנוֹשׁ מָעֹז בְּיוֹם יֹאבַד
כָּל חַי וְיֶעְתַּק צוּר וְהַר נוֹפֵל.

קפא
נָרוּץ לְהִלָּחֵם בְּצִבְאוֹת חֵטְא
תָּמִיד וְאַל נֶחְדַּל עֲדֵי נִירָם
עֹל תַּאֲוֹת נֹעַר נְהִי מַשְׁבִּית
מֶנּוּ וְנָסִיר מַמְלְכוּת נִירָם
כִּי הַנְּשָׁמוֹת כַּאֲדָמוֹת מָלְ-
אוּ קוֹץ וּמַעְשֵׂה הָאֱמֶת נִירָם.

קפב
כַּמָּה יְפַתֵּנִי זְמָן בִּכְלִי
יַיְן וּפַת שֶׁקֶר וּטוּב נָזִיד
נָזִיד וְנֵעָנֵשׁ עֲלֵי מִדּוֹ-
תֵינוּ וּבַדָּבָר אֲשֶׁר נָזִיד.

קפג
נַפְשׁוֹת חֲסִידִים פֵּרְקוּ נִזְמֵי עֲדִי
תֵבֵל וְעֶדְיָהּ מֵעֲלֵיהֶם נִצְּלוּ
כִּי נִבְדְּלוּ מֵחֹטְאִים וַיִּהְיוּ
אֵלֶּה לְדֵרָאוֹן וְאֵלֶּה נִצְּלוּ.

קפד
אֶל מִי בְּיוֹם הַדִּין נְהִי בֹרְחִים
מִצָּר וּמַחְסֵנוּ בְמִי נָשִׂים
אֵין טוֹב לְבַד לָרוּץ לְמִגְדַּל עֹז
הָאֵל וְשָׁמָּה נִהְיֶה נָסִים.

קפה
צִצֵּי חֲסָדִים לָךְ אֱרֵה גַּם הִתְ-
עַנֵּג בְּבָשְׂמֵיהֶם וְטוּב נִרְדָּם
בָּם תִּמְצְאָה טַעַם וְתִתְעַדֵּן
כִּי הֵם יְחַיּוּ לֵב אֱנוֹשׁ נִרְדָּם.

קפו
רַבַּת צְרָרוּנִי בְנֵי יָמִים
גַּם צָרֲרוּ אֹתִי בְמוֹ נִכְלָם
צוּרִי הֲמֹל עַל דַּל וְיוֹם דִּינְךָ
אַל נָא יְהִי שָׁב מִמְּךָ נִכְלָם.

קפז
אִם אֶסְבְּלָה מָצוֹק אֲקַוֶּה כִּי
רִיבִי וּמִשְׁפָּטִי לְאֵל נָגַע
יִרְפָּא נְגָעַי מִזְּמָן אַכְזָר
עַל פִּיו יְהִי כָל רִיב וכָל נָגַע.

קפח
אֱנוֹשׁ יִגְאֶה בְּרֹב עָשְׁרוֹ וְיַרְאֶה
לְחֶבְרַת אֹהֲבָיו כָּל בֵּית נְכֹתֹה
וְיוֹם שִׁבְרוֹ מְזוֹרוֹ לֹא יְרַפֵּא
מְזוֹר רֹפֵא וְצָרְיוֹ עִם נְכֹאתֹה.

קפט
טִירַת שְׂשׂוֹנַי עַל עֲוֹנַי חָרְבָה
וּבְרֹב יְגוֹנוֹתַי כְּמִדְבָּר נִצְּתָה
עָלַי שְׁאֵרִית רַחֲמֵי אֵל חֹפְפוּ
עַתָּה חֲמָתוֹ בִי כְּמוֹ אֵשׁ נִצְּתָה.

קצ
אֲשֶׁר יִזְכּוּ רְאוֹת נֹעַם כְּבוֹד אֵל
בְּבֵיתוֹ הֵם פְּקִידִים נֶאֱמָנִים
וּפִתְחוֹן פֶּה יְהִי לָהֶם בְּיוֹם דִּין
בְּיוֹם הָסִיר שְׂפַת כָּל נֶאֱמָנִים.

קצא
קָרוֹב רְצוֹן הָאֵל לְכָל דֹּרְשָׁיו
נָרוּץ לְהַשּׂיגוֹ וְגַם נָחוּשׁ
טֶרֶם בְּכוֹר מָוֶת בְּשָׂרֵנוּ
יֹאכַל וְאִלּוּ יִהְיֶה נָחוּשׁ.

קצב
אֲשֶׁר רָווּ בְּיֵין תַּאְוָה לְשָׁכְרָה
יְקַוּוּ טוֹב וְתַגְמוּל הַנְּזִרִים
וְהַמָּוֶת יְעוֹלֵל אֶת הֲמוֹנָם
כְּסַלְסִלּוֹת וְיִלְקֹט הַנְּזִירִים.

קצג
אֲשֶׁר נִפְתּוּ בְּתַעְנוּג הַזְּמַנִים
וְצוּף טַעְמָם וְיִשְׁתּוּ יֵין נְסִיכָם
שְׁאוֹל יֵרְדוּ וְשָׁכְבוּ רֹאשׁ וְזָנָב
וְהִשְׁתַּוָּה בְזוּיָם עִם נְסִיכָם.

קצד
עוּרָה חֲצוֹת לַיִל עֲמוֹד מוּל אֵל
עַד לִבְּךָ נֶגְדּוֹ יְהִי נִשְׁפָּךְ
וּבְעֵת מְאוֹרִים יֶחְשְׁכוּ יַחַד
הָאֵר בְּטוּב הַמַּעֲשִׂים נִשְׁפָּךְ.

קצה
מַה תִּפְצְרוּ בִזְמָן לְהִתְיַשּׁר
וְאֵיךְ יַהְפֹּךְ חֲבַרְבֻּרוֹת לְעוֹלָם נָמֵר
לָכֵן מְתֵי סוֹדִי לְכוּ עַד נַחֲלִיף
אֹבֵד בְּעֹמֵד גַּם בְּטוֹב רַע נָמֵר.

קצו
תַּעְבֹר בְּתוֹךְ יַמִים וְתַרְבֶּה בִנְ-
דָרִים וְלֹא תַשְׁלִים לְאֵל נִדְרָךְ
מַה תַּעֲשֶׂה בִּהְיוֹת כְּטִיט חוּצוֹת
גּוּפָךְ וְכֶעָפָר יְהִי נִדְרָךְ.

קצז
עֵינַי לְחַטֹּאתַי בְּכוּ תָּמִיד
וּכְמַעֲיָן נַחֲלֵי בְכִי נִבְעוּ
רִגְזוּ לְיוֹם בּוֹ מַעֲשֵׂי כָל אִישׁ
נִגְלוּ וּמַצְפֻּנָיו לְאֵל נִבְעוּ.

קצח
אָדָם לְאוֹי יֻלַּד וְגַם נִקְרָא
כִּי הוּא לְקֹרוֹת הַזְּמָן נִקְרָה
מָחָר יְדֵי מָוֶת יְבִיאוּהוּ
בִּנְקִיק סְעִיף סֶלַע וְגַם נִקְרָה.

קצט
כָּל אִישׁ אֲשֶׁר לִבּוֹ בְּהוֹד תֵּבֵל
יַחְשֹׁק וּפִיהָ יִהְיֶה נֹשֵׁק
נַפְשׁוֹ יְהִי הֹרֵג בְּלִי שֶׁלַח
רַק תַּאֲוָתוֹ הִיא כְלִי נֹשֵׁק.

ר
נַפְשִׁי בְּגוּף תּוֹךְ מַאֲסָר יֹשָׁבֶת
תִּפְתִּי בְּרוּחַ ליק… נֹשָׁבֶת
גּוּפֵךְ כְּעִיר נֹשְׁבָה בְּיוֹם אֶחָד אֲבָל
מָחָר תְּהִי חָרְבָּה בְּלִי נֹשַׁבֶת.

רא
עַד אָן נְהִי יֹרְדִים שְׁאוֹל מָטָה
אָכֵן וְלִמְרוֹם הַיְקָר נָעַל
יִפְתַּח לְשֹׁכְנָיו שַׁעֲרֵי מָוֶת
אַךְ שַׁעֲרֵי חַיִּים בְּאַף נָעָל.

רב
אַל תָּשֶׁת לֵב אֶל הַחֹטְאִים
שֹׁקְטִים גַּם לֹא אֶל נִיד בָּרִים
כִּי כָל הַמַּעְשִׂים לִפְנֵי אֵל
הֵם נִזְכָּרִים גַּם נִדְבָּרִים.

רג
לוּלֵי יְקוֹד לִבִּי אֲשֶׁר יַתִּיךְ
דִּמְעִי מְקוֹר עֵינַי יְהִי נִקְפָּה
עַל רֹב חֲטָאַי מַק אֲשַׁו בָּשְׂמִי
תַּחַת חֲגוֹרָה אֶעֱשֶׂה נִקְפָּה.
